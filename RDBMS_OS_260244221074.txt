create table salesman(salesman_id int primary key, sname varchar(20), scity varchar(20), commission double);

 create table Customer(customer_id int primary key, cust_name varchar(20), city varchar(20), grade varchar(20), salesman_id int , foreign key(salesman_id) references salesman(salesman_id));

drop procedure if exits sp1;
DELIMITER $$
CREATE PROCEDURE sp1() 
BEGIN
select * from salesman;
update salesman 
set commission = commission +(commission*0.5)
where city = "pune";
end 
$$
